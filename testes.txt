@app.route("/teste")
def base():
    map = folium.Map(
        location=[-22.522814, -44.123209], #mapa inicial
        zoom_start=15
    )

    for nome, coord in locais.items():
        folium.Marker(
            location=coord,
            popup=f"<b>{nome}</b>Lat: {coord[0]:.6f}<br>Lon: {coord[1]:.6f}",
            tooltip=nome,
            icon=folium.Icon(icon="5", color="red", prefix='1')
        ).add_to(map)

    return map._repr_html_()
